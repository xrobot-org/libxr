file(GLOB ${PROJECT_NAME}_DRIVER_SOURCES CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_LIST_DIR}/*.cpp"
)
list(SORT ${PROJECT_NAME}_DRIVER_SOURCES)

target_sources(${PROJECT_NAME} PRIVATE ${${PROJECT_NAME}_DRIVER_SOURCES})

if(CONFIG_IDF_TARGET_ARCH_XTENSA OR
   IDF_TARGET STREQUAL "esp32" OR
   IDF_TARGET STREQUAL "esp32s2" OR
   IDF_TARGET STREQUAL "esp32s3")
  if(NOT DEFINED LIBXR_FASTCOPY_POLICY OR "${LIBXR_FASTCOPY_POLICY}" STREQUAL "")
    set(LIBXR_FASTCOPY_POLICY STD_LIBC)
  endif()
  if(NOT DEFINED LIBXR_FASTSET_POLICY OR "${LIBXR_FASTSET_POLICY}" STREQUAL "")
    set(LIBXR_FASTSET_POLICY STD_LIBC)
  endif()
# 仅在 ESP-IDF CMake 组件流程导出 CONFIG_LIBC_OPTIMIZED_MISALIGNED_ACCESS 时自动切到
# STD_LIBC；预编译库等未注入 CONFIG_* 变量的集成会安全回退至 SHIFTED_ODD / BUILTIN，
# 也可通过 -DLIBXR_FAST*_POLICY 显式指定。
# Auto-switch to STD_LIBC when ESP-IDF CMake injects CONFIG_*; standalone builds
# without CONFIG_* safely fall back to SHIFTED_ODD / BUILTIN (or explicit overrides).
elseif(CONFIG_LIBC_OPTIMIZED_MISALIGNED_ACCESS)
  if(NOT DEFINED LIBXR_FASTCOPY_POLICY OR "${LIBXR_FASTCOPY_POLICY}" STREQUAL "")
    set(LIBXR_FASTCOPY_POLICY STD_LIBC)
  endif()
  if(NOT DEFINED LIBXR_FASTMOVE_POLICY OR "${LIBXR_FASTMOVE_POLICY}" STREQUAL "")
    set(LIBXR_FASTMOVE_POLICY STD_LIBC)
  endif()
endif()
